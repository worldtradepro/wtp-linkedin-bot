        function rangeDates() {
            // Trade Flows: one rolling 7-day window picked on the timeline (matches the server's lane-weeks windows).
            if (state.reportType === 'flow_distortion') {
                var wb = new Date(), wa = new Date();
                wb.setDate(wb.getDate() - 7 * state.weekK);
                wa.setDate(wa.getDate() - 6 - 7 * state.weekK);
                return { from: fmtDate(wa), to: fmtDate(wb) };
            }
            var to = new Date();
            // "Last 7d" stops at day -6, not -7: day -7 belongs to the free "7-15 days"
            // window (wtp_opps_query's free_to), so starting the locked range there would
            // share a day with the free one and leak one real day of paid data through a
            // locked button. "15 days+" starts at -16 so it does not repeat the 7-15 window's days.
            var d6 = new Date(); d6.setDate(d6.getDate() - 6);
            var d7 = new Date(); d7.setDate(d7.getDate() - 7);
            var d15 = new Date(); d15.setDate(d15.getDate() - 15);

            if (state.range === '15plus') return { from: '2020-01-01', to: fmtDate(d16) };
            var from = new Date();
            from.setDate(from.getDate() - parseInt(state.range, 10));
            return { from: fmtDate(from), to: fmtDate(to) };

        function resolveCentroid(country) {
            if (!country) return null;
            var c = country.trim();
            if (CENTROIDS[c]) return CENTROIDS[c];
            if (CENTROIDS_CI[c.toLowerCase()]) return CENTROIDS_CI[c.toLowerCase()];
            var parts = c.split(/,|\/|\band\b|&/i).map(function(p){return p.trim();}).filter(Boolean);
            for (var i = 0; i < parts.length; i++) {
                var p = parts[i];
                if (CENTROIDS[p]) return CENTROIDS[p];
                if (CENTROIDS_CI[p.toLowerCase()]) return CENTROIDS_CI[p.toLowerCase()];
            }
            return null;
        }
        function hasCoords(it) {

        function isMappable(it) {
            // Generalized across report types: EPC uses project|company_hq|global|unknown,
            // Signals use explicit|chokepoint|unknown — either way "unknown" is the only
            // value that means "don't plot this".
            if (!it.location_source || it.location_source === 'unknown') return false;
            // City-level items carry their own real coordinates and don't need a country
            // centroid match at all (covers cases where the model names a city correctly
            // but the country string doesn't match our fixed centroid table's spelling).
            if (hasCoords(it)) return true;
            return !!resolveCentroid(it.country);

        // Sunflower/phyllotaxis spiral: the j-th duplicate pin in the same country is
        // placed at radius ∝ sqrt(j) and angle = j * golden angle around the centroid.
        // Point density this way stays roughly even as count grows (unlike fixed-amplitude
        // random jitter, which just makes a denser and denser blob for busy countries like
        // the US), and neighboring points never coincide since the golden angle is
        // irrational relative to any small integer step.
        var GOLDEN_ANGLE_DEG = 137.50776;
        function spiralOffset(index, baseRadiusDeg) {
            if (index === 0) return { dLat: 0, dLng: 0 };
            var r = baseRadiusDeg * Math.sqrt(index);
            var theta = index * GOLDEN_ANGLE_DEG * Math.PI / 180;
            return { dLat: r * Math.cos(theta), dLng: r * Math.sin(theta) };
        }
        // The radii passed to spiralOffset() below were tuned by eye at "zoomed into a
        // country" altitude (~0.5). At the default whole-globe view (altitude 1.3-1.5
        // depending on device - see DEFAULT_ALTITUDE) the same degree offsets cover
        // only a few screen pixels, so duplicate pins still
        // look stacked until the user manually zooms in. Scale the radius with altitude
        // (apparent pixel size of a fixed degree offset shrinks roughly ∝ 1/altitude for
        // globe.gl's perspective camera) so separation stays visible at whatever zoom
        // level is current; clamped so extreme zoom-out doesn't spiral points into
        // neighboring countries and extreme zoom-in doesn't collapse them back down.
        var ALT_REF = 0.5;
        function altScale() {
            return Math.max(0.5, Math.min(currentAltitude / ALT_REF, 6));
        }

        // ── Lane hit-testing (Trade Flows mode) ───────────────────────────────────
        // Each lane is densified along great circles once, then on every mouse move the
        // visible points are projected to the screen and the cursor's distance to the
        // polyline decides the hit. A lane is picked within LANE_HIT_PX, and once picked it
        // is KEPT until the cursor is more than LANE_KEEP_PX away, so the card does not
        // flicker shut while the cursor drifts along or near the line.
        var LANE_HIT_PX = 16, LANE_KEEP_PX = 36, LANE_TAP_PX = 24;
        function llVec(lat, lng) {
            var a = lat * Math.PI / 180, b = lng * Math.PI / 180;
            return [Math.cos(a) * Math.cos(b), Math.cos(a) * Math.sin(b), Math.sin(a)];
        }
        function densifyLane(pts) {
            var out = [];
            for (var i = 0; i < pts.length - 1; i++) {
                var v0 = llVec(pts[i][0], pts[i][1]), v1 = llVec(pts[i + 1][0], pts[i + 1][1]);
                var dot = Math.max(-1, Math.min(1, v0[0] * v1[0] + v0[1] * v1[1] + v0[2] * v1[2]));
                var ang = Math.acos(dot), steps = Math.max(1, Math.ceil(ang * 180 / Math.PI / 2));
                for (var s = 0; s < steps; s++) {
                    var t = s / steps, sa = Math.sin(ang);
                    var k0 = sa < 1e-6 ? 1 - t : Math.sin((1 - t) * ang) / sa, k1 = sa < 1e-6 ? t : Math.sin(t * ang) / sa;
                    var x = k0 * v0[0] + k1 * v1[0], y = k0 * v0[1] + k1 * v1[1], z = k0 * v0[2] + k1 * v1[2];
                    out.push({ lat: Math.asin(Math.max(-1, Math.min(1, z))) * 180 / Math.PI, lng: Math.atan2(y, x) * 180 / Math.PI, v: [x, y, z] });
                }
            }
            var last = pts[pts.length - 1];
            out.push({ lat: last[0], lng: last[1], v: llVec(last[0], last[1]) });
            return out;
        }
        TRADE_LANES.forEach(function(l) {
            l.dense = densifyLane(l.pts);
            // Lane length in degrees of arc (great-circle sum), used to give every lane the same dash size and speed.
            var len = 0;
            for (var i = 0; i < l.pts.length - 1; i++) {
                var a = llVec(l.pts[i][0], l.pts[i][1]), b = llVec(l.pts[i + 1][0], l.pts[i + 1][1]);
                len += Math.acos(Math.max(-1, Math.min(1, a[0] * b[0] + a[1] * b[1] + a[2] * b[2]))) * 180 / Math.PI;
            }
            l.lenDeg = Math.max(len, 1);
        });
        function segDist(px, py, ax, ay, bx, by) {
            var dx = bx - ax, dy = by - ay, len2 = dx * dx + dy * dy;
            var t = len2 ? Math.max(0, Math.min(1, ((px - ax) * dx + (py - ay) * dy) / len2)) : 0;
            var cx = ax + t * dx, cy = ay + t * dy;
            return Math.sqrt((px - cx) * (px - cx) + (py - cy) * (py - cy));
        }
        // Distance in CSS px from a client point to every lane (Infinity when none of it is visible).
        function laneDistances(clientX, clientY) {
            var canvas = globeWrapEl.querySelector('canvas');
            if (!canvas) return {};
            var rect = canvas.getBoundingClientRect(), mx = clientX - rect.left, my = clientY - rect.top;
            var pov = world.pointOfView(), cv = llVec(pov.lat, pov.lng), minDot = 1 / (1 + pov.altitude) + 0.02;
            var out = {};
            TRADE_LANES.forEach(function(l) {
                var best = Infinity, prev = null;
                l.dense.forEach(function(p) {
                    var vis = p.v[0] * cv[0] + p.v[1] * cv[1] + p.v[2] * cv[2] > minDot;
                    if (!vis) { prev = null; return; }
                    var sc = world.getScreenCoords(p.lat, p.lng, 0.012);
                    if (prev) best = Math.min(best, segDist(mx, my, prev.x, prev.y, sc.x, sc.y));
                    else best = Math.min(best, Math.sqrt((mx - sc.x) * (mx - sc.x) + (my - sc.y) * (my - sc.y)));
                    prev = sc;
                });
                out[l.id] = best;
            });
            return out;
        }
        function lanesActive() { return state.reportType === 'flow_distortion' && state.showLanes; }
        function pickLane(clientX, clientY, keepId, hitPx) {
            var d = laneDistances(clientX, clientY);
            if (keepId && d[keepId] <= LANE_KEEP_PX) return keepId;
            var best = '', bd = hitPx;
            Object.keys(d).forEach(function(id) { if (d[id] <= bd) { bd = d[id]; best = id; } });
            return best;
        }

        // relying on the fixed 90vh in CSS, so however tall the (now-compacted, see
        // body.home header rules) chrome above it turns out to be, the globe always
        // fills the rest of the screen edge-to-edge on first load - no dead gap below
        // it, no scrolling needed to see the full sphere.
        //
        // Bug fix 2026-09-13: this used to measure root.getBoundingClientRect().top,
        // which is relative to the current scroll position, not the page. On mobile,
        // scrolling the page fires a 'resize' event too (the browser's address bar
        // collapses/expands as you scroll, changing window.innerHeight) - so mid-
        // scroll, this ran with an already-negative "top" plus a shifted innerHeight
        // and computed a wildly-too-tall height, which then blew up the globe's
        // rendered size (more canvas pixels for the same camera altitude). Adding
        // back the current scroll offset converts "top" into a fixed, scroll-
        // independent position in the document, so the computed height stays stable
        // no matter where the page happens to be scrolled when this runs.
        //
        // -16 leaves a small gap at the bottom (see the matching margin-bottom on
        // #wtp-imap-root) so the globe doesn't sit flush against whatever follows it.
        // .wim-globe-wrap's min-height floor - mirrors the CSS (220px normally, or
        // 100vw on mobile - see the "Taller globe box" comment in the mobile media
        // query). A function, not a one-time constant, since it depends on the
        // current window width and needs to stay correct across resize/rotation.
        function globeMinHeight() {
            return window.innerWidth <= 760 ? window.innerWidth : 220;
        }
        function fitMapHeight() {
            var docTop = root.getBoundingClientRect().top + window.pageYOffset;
            var vh = window.visualViewport ? window.visualViewport.height : window.innerHeight;
            var fillTarget = vh - docTop - 16;
            // Short-viewport fix 2026-09-14: on a short mobile viewport, the filter
            // bar + zoom controls + sector bar can together need more room than
            // fillTarget leaves for them - and since this element clips overflow
            // (see overflow:hidden above), giving it a height smaller than what its
            // rows actually need silently cuts the bottom off whichever row runs out
            // of room (the sector bar, in practice). None of those rows can shrink
            // below their own content size in a flex column (that's normal flexbox
            // behavior once .wim-zoom-ctrl - see its own comment - stopped losing
            // that protection to its overflow:hidden), so summing their real
            // rendered heights here tells us the true minimum this element needs.
            // Keeping this a definite height (not min-height) matters too: it's what
            // lets .wim-globe-wrap's flex-grow correctly fill "whatever's left" -
            // tried min-height instead first, but with no definite container size to
            // divide up, the globe's own canvas height fed back into its wrapper's
            // auto height with nothing to correct it, and it stuck at a wrong size.
            // Margins count too (e.g. .wim-zoom-ctrl's margin:10px 0 on mobile,
            // .wim-gated-notice's margin-top) - they're gaps the flex column still
            // has to reserve, even though they're not part of any child's own
            // getBoundingClientRect().height.
            var othersHeight = 0;
            Array.prototype.forEach.call(root.children, function(child) {
                if (child === globeWrapEl || child === gatedNoticeEl) return; // the notice is an overlay, it reserves no space
                var cs = getComputedStyle(child);
                othersHeight += child.getBoundingClientRect().height + parseFloat(cs.marginTop || 0) + parseFloat(cs.marginBottom || 0);
            });
            root.style.height = Math.max(fillTarget, othersHeight + globeMinHeight(), 480) + 'px';
            // Bug fix 2026-09-15: .wim-stats (the sector/stage breakdown card, desktop
            // only - it's display:none on mobile) is absolutely positioned with a
            // hardcoded top:16px in CSS, measured from #wtp-imap-root's own top edge.
            // That assumed the filter bar above it always stayed about the same
            // short height, but the filter bar is a normal in-flow row now (see the
            // "Made unconditional" comment above) whose real height varies with its
            // content (wrapping text, the gated-notice banner appearing) - whenever
            // it's taller than whatever the 16px guess assumed, this card's top
            // overlaps into it instead of clearing it. Re-anchoring it to wherever
            // the globe itself actually starts (which is already correctly
            // positioned below the filter bar via normal flow) keeps it aligned no
            // matter how tall the filter bar ends up being.
            if (statsPanelEl) {
                statsPanelEl.style.top = (globeWrapEl.offsetTop) + 'px';
            }
        }

        function applyFilters() {

        function buildPointsData(mapped) {
            var scale = altScale();
            var countryCounts = {};
            var cityCounts = {};
            return mapped.map(function(it) {
                if (hasCoords(it)) {
                    // Real per-item coordinates — no country-centroid fallback needed.
                    // Still spiral-offset exact city dupes, but with a much tighter radius
                    // than the country-level case since these duplicates are already close.
                    var cityKey = it.city || (it.lat + ',' + it.lng);
                    var cj = cityCounts[cityKey] || 0;
                    cityCounts[cityKey] = cj + 1;
                    var cityOff = spiralOffset(cj, 0.09 * scale);
                    return Object.assign({}, it, { __lat: parseFloat(it.lat) + cityOff.dLat, __lng: parseFloat(it.lng) + cityOff.dLng, __isCluster: false });
                }
                var c = resolveCentroid(it.country);
var __centroidKey = c ? (c[0].toFixed(3) + ',' + c[1].toFixed(3)) : null;
                var ck = __centroidKey || (it.country || '?');
                var j = countryCounts[ck] || 0;
                countryCounts[ck] = j + 1;
                var off = spiralOffset(j, 1.2 * scale);
                return Object.assign({}, it, { __lat: c[0] + off.dLat, __lng: c[1] + off.dLng, __isCluster: false });
            });
        }

        // In cluster mode, any country with >1 dupe becomes one badge instead of a spiral
        // of pins — the country-level clumping problem the spiral only partially solves at
        // whole-globe zoom, and the thing that reads worst on a small phone screen where
        // even the altitude-scaled spiral still needs several closely-spaced taps to
        // resolve individual pins. Countries with just one item, and any item that already
        // carries its own precise city coordinates (never grouped by country to begin
        // with), still go through the normal scatter path via buildPointsData — reusing it
        // rather than duplicating the offset math.
        function buildMapLayers(mapped) {
            if (!clusterMode) return { points: buildPointsData(mapped), clusters: [] };
            var countryGroups = {};
            mapped.forEach(function(it) {
                if (hasCoords(it)) return;
                // Keyed by country AND report type: an Infrastructure item and a Trade Flow
                // item in the same country must never collapse into one mixed badge.
                var ck = (it.country || '?') + '||' + it.report_type;
                (countryGroups[ck] = countryGroups[ck] || []).push(it);
            });
            var countryHasBoth = {};
            Object.keys(countryGroups).forEach(function(k) {
                var c = k.split('||')[0];
                countryHasBoth[c] = (countryHasBoth[c] || 0) + 1;
            });
            var clusters = [];
            var clusteredCountries = {};
            Object.keys(countryGroups).forEach(function(gk) {
                var items = countryGroups[gk];
                if (items.length <= 1) return;
                var ck = gk.split('||')[0];
                var kind = items[0].report_type;
                var c = resolveCentroid(ck);
                // Same country, both report types: nudge the Trade Flow badge aside so the
                // two never sit on the exact same pixel (they are separate clusters now).
                if (countryHasBoth[ck] > 1 && kind === 'flow_distortion') c = [c[0] - 1.5, c[1] + 2.2];
                // Badge tint + hover breakdown come from the same per-(report_type,sector)
                // tally as the sidebar bars, sorted so the most common group leads — a flat
                // accent-colored badge would erase the sector-color signal every individual
                // pin on the map otherwise carries.
                var groups = {};
                items.forEach(function(it) {
                    var key = groupKey(it.report_type, it.sector);
                    var label = (it.report_type === 'flow_distortion' ? 'Trade Flow · ' : '') + (it.sector || 'Other');
                    if (!groups[key]) groups[key] = { label: label, color: pinColor(it), count: 0 };
                    groups[key].count++;
                });
                var breakdown = Object.keys(groups).map(function(k){ return groups[k]; }).sort(function(a,b){ return b.count - a.count; });
                clusters.push({ country: ck, report_type: kind, count: items.length, lat: c[0], lng: c[1], color: breakdown[0].color, breakdown: breakdown, items: items, __isCluster: true });
                clusteredCountries[gk] = true;
            });
            var remaining = mapped.filter(function(it) {
                return hasCoords(it) || !clusteredCountries[(it.country || '?') + '||' + it.report_type];
            });
            return { points: buildPointsData(remaining), clusters: clusters };
        }

        // Screen-space clustering (2026-09-06): buildMapLayers above only merges
        // same-country duplicates - it says nothing about *different* countries whose
        // pins happen to land close together on screen at the current zoom/rotation
        // (small nearby countries at whole-globe zoom, or just a country with enough
        // items that the spiral's own points overlap). This second pass runs after
        // buildMapLayers, projects every point/cluster to actual screen pixels via
        // getScreenCoords(), and greedily merges anything within SCREEN_CLUSTER_PX of
        // each other into one bigger badge - regardless of country. Re-run on every
        // animation frame while the camera moves (see the onZoom handler below) so a
        // merged badge splits back apart correctly once the user zooms/rotates far
        // enough that its members are no longer visually on top of each other.
        var SCREEN_CLUSTER_PX = 34;

        function isFrontFacing(lat, lng) {
            var pov = world.pointOfView();
            var toRad = function(d) { return d * Math.PI / 180; };
            var lat1 = toRad(pov.lat), lng1 = toRad(pov.lng), lat2 = toRad(lat), lng2 = toRad(lng);
            var cosAngle = Math.sin(lat1) * Math.sin(lat2) + Math.cos(lat1) * Math.cos(lat2) * Math.cos(lng2 - lng1);
            return cosAngle > 0.15;
        }

        function screenClusterMerge(entries) {
            var withScreen = [];
            entries.forEach(function(d) {
                var lat = d.__isCluster ? d.lat : d.__lat;
                var lng = d.__isCluster ? d.lng : d.__lng;
                if (!isFrontFacing(lat, lng)) return;
                var pt = world.getScreenCoords(lat, lng, 0.02);
                if (!pt) return;
                withScreen.push({ d: d, lat: lat, lng: lng, x: pt.x, y: pt.y });
            });

            var used = new Array(withScreen.length).fill(false);
            var groups = [];
            for (var i = 0; i < withScreen.length; i++) {
                if (used[i]) continue;
                used[i] = true;
                var group = [withScreen[i]];
                for (var j = i + 1; j < withScreen.length; j++) {
                    if (used[j]) continue;
                    var dx = withScreen[i].x - withScreen[j].x, dy = withScreen[i].y - withScreen[j].y;
                    if (withScreen[i].d.report_type === withScreen[j].d.report_type && Math.sqrt(dx * dx + dy * dy) < SCREEN_CLUSTER_PX) { group.push(withScreen[j]); used[j] = true; }
                }
                groups.push(group);
            }

            return groups.map(function(group) {
                if (group.length === 1) return group[0].d;

                var totalCount = 0;
                var breakdownMap = {};
                var countryNames = {};
                var sumLat = 0, sumLng = 0;
                var mergedItems = [];
                group.forEach(function(g) {
                    var d = g.d;
                    countryNames[d.country || 'Unknown'] = true;
                    sumLat += g.lat; sumLng += g.lng;
                    if (d.__isCluster) {
                        totalCount += d.count;
                        (d.items || []).forEach(function(it) { mergedItems.push(it); });
                        d.breakdown.forEach(function(b) {
                            if (!breakdownMap[b.label]) breakdownMap[b.label] = { label: b.label, color: b.color, count: 0 };
                            breakdownMap[b.label].count += b.count;
                        });
                    } else {
                        totalCount += 1;
                        mergedItems.push(d);
                        var key = (d.report_type === 'flow_distortion' ? 'Trade Flow \u00b7 ' : '') + (d.sector || 'Other');
                        if (!breakdownMap[key]) breakdownMap[key] = { label: key, color: pinColor(d), count: 0 };
                        breakdownMap[key].count += 1;
                    }
                });
                var breakdown = Object.keys(breakdownMap).map(function(k) { return breakdownMap[k]; }).sort(function(a, b) { return b.count - a.count; });
                var names = Object.keys(countryNames);
                var label = names.length === 1 ? names[0] : (names.length + ' locations');
                return {
                    __isCluster: true,
                    report_type: group[0].d.report_type,
                    country: label,
                    count: totalCount,
                    lat: sumLat / group.length,
                    lng: sumLng / group.length,
                    color: breakdown[0].color,
                    breakdown: breakdown,
                    items: mergedItems
                };
            });
        }

        var currentCountryLayers = { points: [], clusters: [] };

        function refreshScreenClusters() {
            var combined = currentCountryLayers.points.concat(currentCountryLayers.clusters);
            var merged = screenClusterMerge(combined);
            // Paint order by DOM order (no z-index: a positioned z-index here leaks above the side panel).
            var rank = function(d) { var t = d.__isCluster ? (d.report_type === 'flow_distortion' ? maxTier(d.items) : null) : (isFlow(d) ? flowTier(d) : null); return t === 'crit' ? 2 : (t === 'elev' ? 1 : 0); };
            merged.sort(function(a, b) { return rank(a) - rank(b); });
            world.htmlElementsData(merged);
        }

        // Which signals sit on which lane, from the items the current filters leave (lane
        // filter excluded, so picking one lane does not blank every other lane).
        function computeLaneStates(items) {
            var states = {};
            TRADE_LANES.forEach(function(l) { states[l.id] = { lane: l, items: [], counts: { crit: 0, elev: 0, watch: 0 }, tier: null }; });
            items.forEach(function(it) {
                if (!isFlow(it)) return;
                TRADE_LANES.forEach(function(l) {
                    if (!laneMatches(l, it)) return;
                    var s = states[l.id];
                    s.items.push(it);
                    s.counts[flowTier(it)]++;
                });
            });
            TRADE_LANES.forEach(function(l) {
                var s = states[l.id];
                s.tier = s.counts.crit ? 'crit' : (s.counts.elev ? 'elev' : (s.counts.watch ? 'watch' : null));
                s.items.sort(function(a, b) { return flowImpact(b) - flowImpact(a); });
            });
            return states;
        }
        function updateLanes(items) {
            if (state.reportType !== 'flow_distortion' || !state.showLanes) {
                laneStates = {};
                world.pathsData([]);
                return;
            }
            laneStates = computeLaneStates(items);
            world.pathsData(TRADE_LANES.map(function(l) {
                var s = laneStates[l.id];
                return { id: l.id, name: l.name, pts: l.pts, tier: s.tier, n: s.items.length, len: l.lenDeg };
            }));
        }
        function laneTipContent(id) {
            var s = laneStates[id];
            if (!s) return '';
            var l = s.lane;
            var html = '<span style="color:var(--accent);font-size:10px;font-weight:700;text-transform:uppercase;letter-spacing:.04em">Trade lane</span><br><b>' + escapeHtml(l.name) + '</b><br>' +
                '<span style="color:var(--text-faint)">' + escapeHtml(l.flow) + '</span><br>';
            if (!s.items.length) {
                return html + '<span style="color:var(--text-faint)">No signals on this lane in the current range.</span>';
            }
            html += TIER_ORDER.filter(function(t) { return s.counts[t]; }).map(function(t) {
                return '<span class="wim-tier-tag ' + t + '" style="margin-left:0">' + TIER_DEFS[t].label + ' · ' + s.counts[t] + '</span>';
            }).join(' ');
            // Weekly trend for this lane (Critical + Elevated per week, oldest -> newest). Free
            // users only have the current week unlocked, so they get a lock hint instead.
            var wk = timelineList();

        function flashData() {
            var items = allItems.filter(isFlow);
            if (!items.length) return null;
            // Start with the latest day and widen day by day until there are at least 3 real
            // (Elevated+) signals to choose from: on a quiet day the top of a tiny pool is noise.
            var dates = [];
            items.forEach(function(it) { if (dates.indexOf(it.report_date) < 0) dates.push(it.report_date); });
            dates.sort().reverse();
            var pool = items, n = 1;
            for (; n <= dates.length; n++) {
                var win = dates.slice(0, n);
                pool = items.filter(function(it) { return win.indexOf(it.report_date) >= 0; });
                if (pool.filter(function(it) { return flowImpact(it) >= 8; }).length >= 3) break;
            }
            n = Math.min(n, dates.length);
            var latest = dates[0], oldest = dates[n - 1];
            var strong = pool.filter(function(it) { return flowImpact(it) >= 8; });
            var ranked = (strong.length >= 3 ? strong : pool).slice().sort(byImpactDesc);
            var picks = [], seen = {};
            ranked.forEach(function(it) { if (picks.length < 3 && !seen[it.sector]) { picks.push(it); seen[it.sector] = 1; } }); // spread across sectors first
            ranked.forEach(function(it) { if (picks.length < 3 && picks.indexOf(it) < 0) picks.push(it); });
            picks.sort(byImpactDesc);
            return { date: latest, days: n, label: n > 1 ? tlShort(oldest) + ' – ' + tlShort(latest) : tlShort(latest), count: pool.length, tiers: tierCounts(pool), picks: picks };
        }
        function parseScaleM(s) {
            // Real data mostly says Large / Medium / Small rather than an amount; amounts ("USD 500m") are honoured when present.
            var word = { mega: 10000, large: 3000, medium: 300, small: 30 }[(s || '').trim().toLowerCase()];
            if (word) return word;
            var m = /(\d[\d,.]*)\s*(billion|bn|b|million|mn|m|k)?/i.exec(s || '');
            if (!m) return 0;
            var n = parseFloat(m[1].replace(/,/g, '')), u = (m[2] || '').toLowerCase();
            if (u === 'billion' || u === 'bn' || u === 'b') n *= 1000; else if (u === 'k') n /= 1000;
            return isNaN(n) ? 0 : n;
        }
        function projectsData() {
            var items = allItems.filter(function(it) { return !isFlow(it); });
            if (!items.length) return null;
            var bySector = {}, byCountry = {}, minDate = '9999', maxDate = '';
            items.forEach(function(it) {
                bySector[it.sector] = (bySector[it.sector] || 0) + 1;
                if (it.country) byCountry[it.country] = (byCountry[it.country] || 0) + 1;
                if (it.report_date < minDate) minDate = it.report_date;
                if (it.report_date > maxDate) maxDate = it.report_date;
            });
            // Early leads first (S1-S3: not yet at tender - what BD teams can still act on), then by size.
            var early = function(it) { return /^S[123]/.test(it.stage || '') ? 1 : 0; };
            var preTender = items.filter(early).length;
            var ranked = items.slice().sort(function(a, b) { return (early(b) - early(a)) || (parseScaleM(b.scale) - parseScaleM(a.scale)); });
            var picks = [], per = {};
            ranked.forEach(function(it) { if (picks.length < 5 && (per[it.sector] || 0) < 2) { picks.push(it); per[it.sector] = (per[it.sector] || 0) + 1; } }); // max 2 per sector
            ranked.forEach(function(it) { if (picks.length < 5 && picks.indexOf(it) < 0) picks.push(it); });
            var sectors = Object.keys(bySector).sort(function(a, b) { return bySector[b] - bySector[a]; });
            return { from: minDate, to: maxDate, total: items.length, preTender: preTender, bySector: bySector, sectors: sectors, picks: picks };
        }

        function roundRectPath(ctx, x, y, w, h, r) {
            ctx.beginPath();
            ctx.moveTo(x + r, y);
            ctx.arcTo(x + w, y, x + w, y + h, r);
            ctx.arcTo(x + w, y + h, x, y + h, r);
            ctx.arcTo(x, y + h, x, y, r);
            ctx.arcTo(x, y, x + w, y, r);
            ctx.closePath();
        }
        function cardFont(w, s) { return w + ' ' + s + 'px -apple-system, "Segoe UI", Roboto, Arial, sans-serif'; }
        function fitText(ctx, text, maxW) {
            if (ctx.measureText(text).width <= maxW) return text;
            while (text.length > 1 && ctx.measureText(text + '…').width > maxW) text = text.slice(0, -1);
            return text + '…';
        }
        function wrapLines(ctx, text, maxW, maxLines) {
            var words = text.split(/\s+/), lines = [], line = '';
            for (var i = 0; i < words.length; i++) {
                var test = line ? line + ' ' + words[i] : words[i];
                if (ctx.measureText(test).width > maxW && line) {
                    lines.push(line); line = words[i];
                    if (lines.length === maxLines) { line = ''; lines[maxLines - 1] = fitText(ctx, lines[maxLines - 1] + ' ' + words.slice(i).join(' '), maxW); break; }
                } else line = test;
            }
            if (line && lines.length < maxLines) lines.push(fitText(ctx, line, maxW));
            return lines;
        }
        // Globe capture into a rounded square + redraw of the DOM pins. Returns false if the browser will not let us read the WebGL canvas.
        function drawGlobeInto(ctx, GX, GY, GS) {
            try {
                var gc = globeWrapEl.querySelector('canvas');
                world.renderer().render(world.scene(), world.camera()); // a WebGL canvas is only readable right after a render
                var dpr = gc.width / gc.getBoundingClientRect().width;
                var side = Math.min(gc.width, gc.height);
                var sx = (gc.width - side) / 2, sy = (gc.height - side) / 2;
                ctx.save();
                roundRectPath(ctx, GX, GY, GS, GS, 28); ctx.clip();
                ctx.fillStyle = '#0a1730'; ctx.fillRect(GX, GY, GS, GS);
                ctx.drawImage(gc, sx, sy, side, side, GX, GY, GS, GS);
                var k = GS / (side / dpr), offX = sx / dpr, offY = sy / dpr;
                var pov = world.pointOfView(), cvv = llVec(pov.lat, pov.lng), minDot = 1 / (1 + pov.altitude) + 0.02;
                (world.htmlElementsData() || []).forEach(function(d) {
                    var lat = d.__isCluster ? d.lat : d.__lat, lng = d.__isCluster ? d.lng : d.__lng;
                    var v = llVec(lat, lng);
                    if (v[0] * cvv[0] + v[1] * cvv[1] + v[2] * cvv[2] <= minDot) return;
                    var sc = world.getScreenCoords(lat, lng, 0.02);
                    var px = GX + (sc.x - offX) * k, py = GY + (sc.y - offY) * k;
                    if (px < GX + 6 || px > GX + GS - 6 || py < GY + 6 || py > GY + GS - 6) return;
                    var flow = d.__isCluster ? d.report_type === 'flow_distortion' : isFlow(d);
                    var color = flow ? TIER_DEFS[d.__isCluster ? maxTier(d.items) : flowTier(d)].color : (d.__isCluster ? d.color : pinColor(d));
                    var rad = d.__isCluster ? 17 : 11;

                    ctx.beginPath(); ctx.arc(px, py, rad, 0, Math.PI * 2); ctx.fillStyle = color; ctx.fill();

                        ctx.fillText(String(d.count > 99 ? '99+' : d.count), px, py + 6); ctx.textAlign = 'left';
                    }
                });
                ctx.restore();
                gc.toDataURL(); // throws here (not later, on export) if the WebGL canvas is cross-origin tainted
                return true;
            } catch (e) {
                try { ctx.restore(); } catch (e2) {}
                roundRectPath(ctx, GX, GY, GS, GS, 28);
                ctx.fillStyle = '#0a1730'; ctx.fill();

                ctx.fillText('Globe preview unavailable in this browser', GX + GS / 2, GY + GS / 2); ctx.textAlign = 'left';
                return false;
            }

        // The shared host occasionally answers a burst of parallel requests with a transient 503
        // (seen right after a deploy); retrying twice after a short pause turns that into a slower
        // load instead of an empty map.
        function fetchJson(url, retries) {
            return fetch(url).then(function(res) {
                if (!res.ok) throw new Error('HTTP ' + res.status);
                return res.json();
            }).catch(function(err) {
                if (retries > 0) return new Promise(function(done) { setTimeout(done, 900); }).then(function() { return fetchJson(url, retries - 1); });
                throw err;
            });
        }
        function fetchOne(reportType, r) {
            var url = API + '?report_type=' + encodeURIComponent(reportType) + '&from=' + r.from + '&to=' + r.to + '&limit=10000';
            return fetchJson(url, 2);
        }

        // Guards against out-of-order responses: rapid filter clicks fire a new load()
        // before the previous one's fetches resolve, and network timing doesn't guarantee
        // they resolve in the order they were sent — without this, a slow-but-stale
        // response arriving after a newer one can silently overwrite the correct data.
        var loadToken = 0;
        function load() {
            var myToken = ++loadToken;
            var r = rangeDates();
